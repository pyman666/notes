## 强制参数

`/` **之前只能使用位置参数**，`*`**之后的只能使用关键字参数**。在以下的例子中，形参 a 和 b 必须使用指定位置参数，c 或 d 可以是位置形参或关键字形参，而 e 和 f 要求为关键字形参:
```python
def f(a, b, /, c, d, *, e, f):
    pass

f(10, 20, 30, d=40, e=50, f=60)  # 正确
f(10, b=20, c=30, d=40, e=50, f=60)  # b 不能使用关键字参数
f(10, 20, 30, 40, 50, f=60)  # e 必须使用关键字参数
```
## 默认参数

- 默认参数不能是可变类型

- 获取默认参数：
```python
# 函数的参数名和默认值转字典
# 法一
{i.name: i.default for i in inspect.signature(f).parameters.values()}
# 法二
dict(zip(f.__code__.co_varnames, f.__defaults__))
```
- 检查参数传递：一个函数需要测试某个可选参数是否被传递进来，不能用某个默认值比如None、0或者False值来测试（因为这些值都是合法的值，是可能被用户传递进来的）
```python
_no_value = object()
def spam(a, b=_no_value):
    if b is _no_value:
        print('No b value supplied')

>>> spam(1)
No b value supplied
>>> spam(1, None)
>>>
```
- 默认参数的值仅仅在函数**定义的时候赋值一次**
```python
>>> x = 42
>>> def spam(a, b=x):
        print(a, b)

>>> spam(1)
1 42

>>> x = 23 # 不影响
>>> spam(1)
1 42
```
## 匿名函数

lambda表达式中的x是一个自由变量，在**运行时**绑定值，而不是定义时。

在下面的例子中，python中**循环是没有域的**概念的，在列表中添加lambda的时候，并没有保存n的值。同样也可以利用[**闭包**](https://zhuanlan.zhihu.com/p/22229197)来实现。
```python
funcs = [lambda x: x+n for n in range(3)]
for f in funcs:
    print(f(0))
# 输出 2 2 2

funcs = [lambda x, n=n: x+n for n in range(3)]
for f in funcs:
    print(f(0))
# 输出 0 1 2
```
# 闭包

闭包**无法修改**外部函数的局部变量，若要修改，需加上`nonlocal`关键字
```python
def make_handler():
    s = 0
    def handler():
        print(s)  # 访问s
    return hander
make_handler()()  # 0

def make_handler():
    s = 0
    def handler():
        s += 1  # 改变s，说明不是用外层函数的s
        print(s)
    return hander
make_handler()()  # 报错，s未定义。

def make_handler():
    s = 0
    def handler():
        nonlocal s
        s += 1
        print(s)
    return handler
make_handler()()  # 1
```
可以编写访问函数并将其作为函数属性绑定到闭包上
```python
def sample():
    n = 0
    def func():
        print('n=', n)
    def get_n():
        return n
    def set_n(value):
        nonlocal n
        n = value
    # Attach as function attributes
    func.get_n = get_n
    func.set_n = set_n
    return func

f = sample()
f()  # n= 0
f.set_n(10)
f()  # n= 10
f.get_n()  # 10
```

## 函数

### 函数 @ 函数
```python
def decorator(func):
    def wrapper(*args, **kwargs):
        print("func start")
        return func(*args, **kwargs)
    return wrapper

@decorator
def foo():  # 等价于 foo = decorator(foo)
    pass
```
带参数
```python
def decorator(n):
		def inner(func):
		    def wrapper(*args, **kwargs):
		        print(f"func start {'-' * n}")
		        return func(*args, **kwargs)
		    return wrapper
    return inner

@decorator(10)
def foo():  # 等价于 foo = decorator(10)(foo)
    pass
```
究极形态
```python
from functools import wraps, partial
import logging

def logged(func=None, *, level=logging.DEBUG, name=None, msg=None):
    if func is None:
        return partial(logged, level=level, name=name, msg=msg)

    logname = name if name else func.__module__
    log = logging.getLogger(logname)
    logmsg = msg if msg else func.__name__

    @wraps(func)
    def wrapper(*args, **kwargs):
        log.log(level, logmsg)
        return func(*args, **kwargs)

    return wrapper

@logged
def add(x, y):
    return x + y
```
### 函数 @ 类
```python
def decorator(cls):
    def __str__(self):
        return str(self.__str__)

    cls.__str__ = __str__
    return cls

@decorator
class Foo:  # 等价于 Foo = decorator(Foo)
    pass
```
### 函数 @ 类方法
```python
def decorator(func):
    def wrapper(self, *args, **kwargs):
        print(f"func start")
        return func(self, *args, **kwargs)
    return wrapper

class Foo(object):
    def __init__(self):
        pass

    @decorator
    def foo(self, n):
        print('foo')

a = Foo()
a.foo(1)
```
## 类

### 类 @ 函数

类装饰器装饰函数，默认调用`__call__`方法
```python
class Decorator:
    def __init__(self, func):
        self.func = func

    def __call__(self, *args, **kwargs):
     		print(f"func start")
        return self.func(*args, **kwargs)

@Decorator
def foo():  # 等价于 foo = Decorator(foo)
    pass
```
带参数
```python
class Decorator:
    def __init__(self, n):
        self.n = n

    def __call__(self, func):
        def wrapper(*args, **kwargs)
	     		print(f"func start {'-' * self.n}")
	        return func(*args, **kwargs)
        return wrapper

@Decorator(10)
def foo():  # 等价于 foo = Decorator(10)(foo)
    pass
```

### 类方法 @ all
```python
class Decorator:
	  def decorator(func):
		    def wrapper(*args, **kwargs):
		        print("func start")
		        return func(*args, **kwargs)
		    return wrapper

    @decorator
    def foo(self, n):
        return n

    decorator = staticmethod(decorator)  # 在最后设置

ins = Decorator()

@Decorator.decorator
def f():
    pass

@ins.decorator
def g():
    pass
```
## 执行顺序
```python
from functools import wraps

def d1(func):
    @wraps(func)
    def wrapper():
        print("d1-1")
        func()
        print("d1-2")
    return wrapper

def d2(func):
    @wraps(func)
    def wrapper():
        print("d2-1")
        func()
        print("d2-2")
    return wrapper

@d1
@d2
def foo():
    print("f")

>>> foo()
d1-1
d2-1
f
d2-2
d1-2
```
